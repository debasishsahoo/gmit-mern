// This script captures console.log output from each exercise
// and displays it in the corresponding output area.

function runExercise(num) {
    const codeArea = document.getElementById("code" + num);
    const outputArea = document.getElementById("output" + num);
    const code = codeArea.value;

    // Clear previous output
    outputArea.textContent = "";

    // Capture console.log calls
    const logs = [];
    const originalLog = console.log;

    console.log = function() {
        const args = Array.from(arguments).map(function(arg) {
            if (typeof arg === "object" && arg !== null) {
                try {
                    return JSON.stringify(arg);
                } catch (e) {
                    return String(arg);
                }
            }
            return String(arg);
        });
        const line = args.join(" ");
        logs.push(line);
        originalLog.apply(console, arguments);
    };

    try {
        // For Exercise 7, we run in global scope so DOM access works
        if (num === 7) {
            const fn = new Function(code);
            fn();
        } else {
            // For other exercises, wrap in a function to isolate scope
            const fn = new Function(code);
            fn();
        }
    } catch (error) {
        logs.push("RUNTIME ERROR: " + error.message);
        originalLog.error(error);
    }

    // Restore console.log
    console.log = originalLog;

    // Display captured output
    if (logs.length === 0) {
        outputArea.textContent = "(No output. Make sure you are using console.log.)";
    } else {
        outputArea.textContent = logs.join("\n");
    }
}
</script>
